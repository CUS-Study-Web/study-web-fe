// ─── PAGE: Flashcard Detail (Topic) ───────────────────────────────────────────
function FlashcardDetailPage({ navigate, isLoggedIn, onLogout }: { navigate: (p: Page) => void; isLoggedIn?: boolean; onLogout?: () => void }) {
  const topic = FLASHCARD_TOPICS[0]
  type Filter = "all" | "learned" | "unlearned"
  const [filter, setFilter] = useState<Filter>("all")

  const wordsWithStatus = FLASHCARD_WORDS.map((w, i) => ({ ...w, learned: i < 3 }))
  const filtered = filter === "all" ? wordsWithStatus : filter === "learned" ? wordsWithStatus.filter(w => w.learned) : wordsWithStatus.filter(w => !w.learned)

  const pct = Math.round((topic.learned / topic.total) * 100)

  const typeColor: Record<string, { bg: string; color: string }> = {
    Noun:      { bg: "#DDEAF8", color: "#2F6FAE" },
    Adjective: { bg: "#F5E6F3", color: "#9B4E8D" },
    Verb:      { bg: "#DCE9DE", color: "#2C5A31" },
  }

  const filterBtns: { key: Filter; label: string; count: number }[] = [
    { key: "all",      label: "Tất cả",      count: wordsWithStatus.length },
    { key: "learned",  label: "Đã nhớ",      count: wordsWithStatus.filter(w => w.learned).length },
    { key: "unlearned",label: "Chưa thuộc",  count: wordsWithStatus.filter(w => !w.learned).length },
  ]

  return (
    <>
      <NavBar navigate={navigate} currentPage="home" isLoggedIn={isLoggedIn} onLogout={onLogout} />

      {/* Header */}
      <div style={{ background: "linear-gradient(135deg, #2C5A31 0%, #1e4023 100%)", padding: "40px 24px 48px" }}>
        <div style={{ maxWidth: 1000, margin: "0 auto" }}>
          {/* Breadcrumb */}
          <div style={{ display: "flex", alignItems: "center", gap: 6, marginBottom: 18 }}>
            <button onClick={() => navigate("flashcard-topics")}
              style={{ background: "none", border: "none", cursor: "pointer", fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 600, fontSize: 13, color: "rgba(220,233,222,0.7)", padding: 0 }}>
              Flashcard
            </button>
            <svg width="14" height="14" viewBox="0 0 24 24" fill="none"><path d="M9 18l6-6-6-6" stroke="rgba(220,233,222,0.5)" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/></svg>
            <span style={{ fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 600, fontSize: 13, color: "#DCE9DE" }}>{topic.title}</span>
          </div>

          <h1 style={{ fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 900, fontSize: 30, color: "#fff", margin: "0 0 20px", letterSpacing: "-0.5px", lineHeight: 1.25 }}>{topic.title}</h1>

          {/* Stats + progress + button */}
          <div style={{ display: "flex", alignItems: "flex-end", gap: 32, flexWrap: "wrap" }}>
            <div style={{ flex: 1, minWidth: 240 }}>
              <div style={{ display: "flex", justifyContent: "space-between", marginBottom: 8 }}>
                <span style={{ fontFamily: "'Noto Sans', sans-serif", fontSize: 13, color: "rgba(220,233,222,0.8)" }}>Đã nhớ {topic.learned}/{topic.total} từ vựng</span>
                <span style={{ fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 700, fontSize: 13, color: "#F5C518" }}>{pct}%</span>
              </div>
              <div style={{ height: 8, borderRadius: 4, background: "rgba(255,255,255,0.2)", overflow: "hidden" }}>
                <div style={{ height: "100%", width: `${pct}%`, borderRadius: 4, background: "linear-gradient(90deg, #5DBB70, #F5C518)", transition: "width 500ms ease" }} />
              </div>
              <div style={{ display: "flex", gap: 20, marginTop: 14 }}>
                {[
                  { label: "Tổng từ", value: topic.total },
                  { label: "Đã nhớ", value: topic.learned, color: "#5DBB70" },
                  { label: "Chưa thuộc", value: topic.total - topic.learned, color: "#FFCDD2" },
                ].map(s => (
                  <div key={s.label}>
                    <div style={{ fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 800, fontSize: 22, color: s.color ?? "#fff", lineHeight: 1 }}>{s.value}</div>
                    <div style={{ fontFamily: "'Noto Sans', sans-serif", fontSize: 11, color: "rgba(220,233,222,0.65)", marginTop: 3 }}>{s.label}</div>
                  </div>
                ))}
              </div>
            </div>
            <div style={{ display: "flex", flexDirection: "column", alignItems: "flex-end", gap: 6 }}>
              <button onClick={() => navigate("flashcard-study")}
                style={{ fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 800, fontSize: 15, padding: "14px 36px", borderRadius: 14, border: "none", background: "#F5C518", color: "#1B1F1C", cursor: "pointer", boxShadow: "0 4px 20px rgba(0,0,0,0.3)", transition: "all 140ms", display: "flex", alignItems: "center", gap: 8 }}
                onMouseEnter={e => { (e.currentTarget as HTMLButtonElement).style.transform = "translateY(-2px)" }}
                onMouseLeave={e => { (e.currentTarget as HTMLButtonElement).style.transform = "" }}>
                <svg width="16" height="16" viewBox="0 0 24 24" fill="none"><path d="M5 3l14 9-14 9V3z" fill="#1B1F1C"/></svg>
                Luyện tập
              </button>
              <span style={{ fontFamily: "'Noto Sans', sans-serif", fontSize: 10, color: "rgba(220,233,222,0.45)", fontStyle: "italic" }}>💡 Prototype: chuyển đến trang học thẻ</span>
            </div>
          </div>
        </div>
      </div>

      {/* Body */}
      <div style={{ maxWidth: 1000, margin: "-20px auto 0", padding: "0 24px 80px" }}>
        <div style={{ background: "#fff", borderRadius: 20, boxShadow: "0 2px 16px rgba(0,0,0,0.07)", border: "1px solid #E4EBE5", overflow: "hidden" }}>

          {/* Filter row */}
          <div style={{ padding: "16px 22px", borderBottom: "1px solid #F0F4F1", display: "flex", gap: 8, alignItems: "center" }}>
            {filterBtns.map(f => (
              <button key={f.key} onClick={() => setFilter(f.key)}
                style={{
                  fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 700, fontSize: 12,
                  padding: "6px 16px", borderRadius: 999, cursor: "pointer", transition: "all 140ms",
                  border: filter === f.key ? "none" : "1.5px solid #D4DCD5",
                  background: filter === f.key ? "#2C5A31" : "#fff",
                  color: filter === f.key ? "#fff" : "#6B746D",
                  boxShadow: filter === f.key ? "0 2px 8px rgba(44,90,49,0.25)" : "none",
                }}>
                {f.label} <span style={{ opacity: 0.7, marginLeft: 3 }}>({f.count})</span>
              </button>
            ))}
            <div style={{ marginLeft: "auto", fontFamily: "'Noto Sans', sans-serif", fontSize: 12, color: "#A0AAA2" }}>{filtered.length} từ vựng</div>
          </div>

          {/* Table */}
          <table style={{ width: "100%", borderCollapse: "collapse" }}>
            <thead>
              <tr style={{ background: "#F4F7F4" }}>
                {["Tiếng Anh", "Phiên âm", "Từ loại", "Nghĩa tiếng Việt", "Trạng thái"].map((h, i) => (
                  <th key={h} style={{ padding: "12px 18px", fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 700, fontSize: 11, color: "#6B746D", textAlign: i === 4 ? "center" : "left", textTransform: "uppercase", letterSpacing: "0.4px", whiteSpace: "nowrap", borderBottom: "1px solid #E4EBE5" }}>{h}</th>
                ))}
              </tr>
            </thead>
            <tbody>
              {filtered.length === 0 ? (
                <tr><td colSpan={5} style={{ padding: "48px", textAlign: "center", fontFamily: "'Noto Sans', sans-serif", fontSize: 14, color: "#A0AAA2" }}>Không có từ vựng nào.</td></tr>
              ) : filtered.map(w => {
                const badge = typeColor[w.type] ?? { bg: "#F4F7F4", color: "#6B746D" }
                return (
                  <tr key={w.id}
                    style={{ borderBottom: "1px solid #F4F7F4", transition: "background 120ms" }}
                    onMouseEnter={e => { (e.currentTarget as HTMLTableRowElement).style.background = "#FAFCFA" }}
                    onMouseLeave={e => { (e.currentTarget as HTMLTableRowElement).style.background = "" }}>
                    <td style={{ padding: "14px 18px" }}>
                      <span style={{ fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 800, fontSize: 15, color: "#1B1F1C" }}>{w.word}</span>
                    </td>
                    <td style={{ padding: "14px 18px" }}>
                      <span style={{ fontFamily: "'Noto Sans', sans-serif", fontSize: 13, color: "#6B746D", fontStyle: "italic" }}>{w.phonetic}</span>
                    </td>
                    <td style={{ padding: "14px 18px" }}>
                      <span style={{ background: badge.bg, color: badge.color, borderRadius: 999, padding: "3px 10px", fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 700, fontSize: 11 }}>{w.type}</span>
                    </td>
                    <td style={{ padding: "14px 18px" }}>
                      <span style={{ fontFamily: "'Noto Sans', sans-serif", fontSize: 14, color: "#3D4540" }}>{w.meaning}</span>
                    </td>
                    <td style={{ padding: "14px 18px", textAlign: "center" }}>
                      {w.learned
                        ? <span style={{ background: "#DCE9DE", color: "#2C5A31", borderRadius: 999, padding: "4px 12px", fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 700, fontSize: 11, whiteSpace: "nowrap", display: "inline-flex", alignItems: "center", gap: 4 }}>
                            <svg width="10" height="10" viewBox="0 0 12 12" fill="none"><circle cx="6" cy="6" r="5" fill="#2C5A31"/><path d="M3.5 6l2 2 3-3" stroke="#fff" strokeWidth="1.3" strokeLinecap="round" strokeLinejoin="round"/></svg>
                            Đã nhớ
                          </span>
                        : <span style={{ background: "#F4F7F4", color: "#6B746D", borderRadius: 999, padding: "4px 12px", fontFamily: "'Be Vietnam Pro', sans-serif", fontWeight: 700, fontSize: 11, whiteSpace: "nowrap" }}>Chưa thuộc</span>
                      }
                    </td>
                  </tr>
                )
              })}
            </tbody>
          </table>
        </div>
      </div>
    </>
  )
}
